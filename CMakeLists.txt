cmake_minimum_required(VERSION 3.14...3.28)
project(AgrillaMoE VERSION 1.0.0 LANGUAGES C CXX)

# AgrillaMoE — server di inferenza dedicato a Qwen3.6-35B-A3B (GGUF Unsloth)
# basato sul fork llama.cpp "moe-expansion" (github.com/vagrillo/llama.cpp).
#
# Collega la libreria llama-server-impl del fork e aggiunge:
#   - selezione interattiva del quant Unsloth (proposta in base alla VRAM)
#   - download automatico via CLI `hf` quando il modello manca
#   - profilo MoE-expansion di default del benchmark RUN1209 (Q8_0, GPQA-D 84.34%)
#   - bind di default 127.0.0.1:8071 e apertura automatica del browser

# ---- individua l'albero llama.cpp (fork moe-expansion) ----
set(AGRILLA_LLAMA_DIR "" CACHE PATH "Cartella del fork llama.cpp (branch moe-expansion)")
if(NOT AGRILLA_LLAMA_DIR)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../repo/tools/server/CMakeLists.txt")
        set(AGRILLA_LLAMA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../repo")
    elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/tools/server/CMakeLists.txt")
        set(AGRILLA_LLAMA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp")
    else()
        message(FATAL_ERROR
            "llama.cpp (branch moe-expansion) non trovato. Esegui ./bootstrap-llama.sh nella cartella AgrillaMoE, oppure passa -DAGRILLA_LLAMA_DIR=/percorso/di/llama.cpp")
    endif()
endif()
message(STATUS "AgrillaMoE: uso llama.cpp da ${AGRILLA_LLAMA_DIR}")

# ---- default di build (overridabili da riga di comando: niente FORCE) ----
if(NOT DEFINED BUILD_SHARED_LIBS)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "build statico")
endif()
if(NOT DEFINED GGML_CUDA)
    set(GGML_CUDA ON CACHE BOOL "backend CUDA")
endif()
if(NOT DEFINED GGML_STATIC)
    set(GGML_STATIC ON CACHE BOOL "link statico cudart/cublas")
endif()
if(NOT DEFINED GGML_BACKEND_DL)
    set(GGML_BACKEND_DL OFF CACHE BOOL "backend incorporati")
endif()
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES "native" CACHE STRING "arch CUDA (es. 61, 75;86, native)")
endif()
if(NOT DEFINED CMAKE_CUDA_RUNTIME_LIBRARY)
    set(CMAKE_CUDA_RUNTIME_LIBRARY Static CACHE STRING "cudart statico")
endif()
if(NOT DEFINED LLAMA_CURL)
    set(LLAMA_CURL OFF CACHE BOOL "niente dipendenza libcurl")
endif()
if(NOT DEFINED LLAMA_OPENSSL)
    set(LLAMA_OPENSSL OFF CACHE BOOL "niente dipendenza openssl")
endif()
if(NOT DEFINED LLAMA_BUILD_IS_DEV)
    set(LLAMA_BUILD_IS_DEV OFF CACHE BOOL "release")
endif()
if(NOT DEFINED LLAMA_BUILD_COMMON)
    set(LLAMA_BUILD_COMMON ON CACHE BOOL "")
endif()
if(NOT DEFINED LLAMA_BUILD_TOOLS)
    set(LLAMA_BUILD_TOOLS ON CACHE BOOL "")
endif()
if(NOT DEFINED LLAMA_BUILD_SERVER)
    set(LLAMA_BUILD_SERVER ON CACHE BOOL "")
endif()
if(NOT DEFINED LLAMA_BUILD_TESTS)
    set(LLAMA_BUILD_TESTS OFF CACHE BOOL "")
endif()
if(NOT DEFINED LLAMA_BUILD_EXAMPLES)
    set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "")
endif()
if(NOT DEFINED LLAMA_BUILD_APP)
    set(LLAMA_BUILD_APP OFF CACHE BOOL "")
endif()
if(NOT DEFINED CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
if(MSVC AND NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded" CACHE STRING "runtime MT statico")
endif()

add_subdirectory("${AGRILLA_LLAMA_DIR}" llama.cpp EXCLUDE_FROM_ALL)

add_executable(agrillamoe src/main.cpp)
target_compile_features(agrillamoe PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(agrillamoe PRIVATE /utf-8)
endif()
target_link_libraries(agrillamoe PRIVATE llama-server-impl)
if(WIN32)
    target_link_libraries(agrillamoe PRIVATE shell32 ws2_32)
endif()

install(TARGETS agrillamoe RUNTIME)
